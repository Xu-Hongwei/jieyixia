        await other.evaluate(async value => {
          const db = await import('/src/services/storage/db.js')
          const p = (await db.allDocs()).records.find(d=>d.id==='review-plan-a')
          await db.putDoc({...p,planned_minutes:value})
        },value)
        if(sync) await page.evaluate(async()=>{await (await import('/src/stores/app.js')).useAppStore().reloadDocs()})
      }
      const btn = name=>page.getByRole('button',{name,exact:true})
      const editor = ['edit','move'].includes(op)
      await btn({edit:'编辑',move:'改期',cancel:'取消安排',progress:'确认进度'}[op]).click()
      if(op==='edit') await page.getByLabel('工作说明',{exact:true}).fill('双页修改的草稿')
      if(op==='move') {
        await btn('安排日期').click()
        await btn('2026-11-24').click()
      }
      await patch(60)
      if(editor) await btn(op==='edit'?'保存修改':'确认改期').click()
      else await page.getByRole('alertdialog').getByRole('button',{name:op==='cancel'?'取消安排':'确认进度',exact:true}).click()

      results.push({probe:'two-pages-repeated-change',op,sync,pass:true,beforeConfirm,afterConfirm})
      await context.close()

    results.push({probe:'multiple-fields-only-first-displayed',shown,minutesShown:shown.includes('90'),minutesBefore:90,minutesAfter:after.planned_minutes,textAfter:after.action_text,dialogs:await page.getByRole('alertdialog').count()})
